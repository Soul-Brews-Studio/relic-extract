#!/bin/zsh
# the 15m gb300 lane for at most 3 days: run now until 08:00, then each night 23:00-08:00, until it finishes.
# readings.py saves its cache on SIGTERM, so each run resumes where the last one stopped.
export PATH=$HOME/.local/bin:/opt/homebrew/bin:$PATH
cd /opt/Code/github.com/Soul-Brews-Studio/relic-extract || exit 1
END=$(( $(date +%s) + 3 * 86400 ))
at() { local t=$(date -j -f '%Y-%m-%d %H:%M' "$(date +%Y-%m-%d) $1" +%s); (( t <= $(date +%s) )) && t=$(( t + 86400 )); echo $t; }
T0=$(date +%s); N=0
while (( $(date +%s) < END )); do
  N=$(( N + 1 )); STOP=$(at 08:00); (( STOP > END )) && STOP=$END
  echo "== run $N from $(date +%H:%M), stops at $(date -r $STOP '+%a %H:%M')"
  .venv/bin/python report/readings.py --unit 15m --model gb300:qwen3:14b --deep --min 3 &
  PID=$!
  ( sleep $(( STOP - $(date +%s) )); kill -TERM $PID 2>/dev/null ) &
  DOG=$!
  wait $PID; RC=$?
  kill $DOG 2>/dev/null
  if (( RC == 0 )); then echo "== gb300 15m deep done in $(( $(date +%s) - T0 ))s over $N runs"; exit 0; fi
  echo "== run $N stopped (rc=$RC) at $(date +%H:%M)"
  NEXT=$(at 23:00); (( NEXT >= END )) && break
  echo "== waiting until $(date -r $NEXT '+%a %H:%M')"
  sleep $(( NEXT - $(date +%s) ))
done
echo "== 3 days over, lane not finished (cache kept; run again to resume)"; exit 3
(exit 0 · 0.0 s)
